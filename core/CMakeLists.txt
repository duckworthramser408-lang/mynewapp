cmake_minimum_required(VERSION 3.22)
project(g23 LANGUAGES CXX)
add_library(g23 SHARED src/g23.cpp src/lut.cpp)
target_include_directories(g23 PUBLIC include)
target_compile_features(g23 PUBLIC cxx_std_17)
target_compile_options(g23 PRIVATE -O2 -ffp-contract=off -Wall -Wextra -Wpedantic)
option(G23_TESTS "Build host tests" OFF)
if(G23_TESTS)
 enable_testing()
 add_executable(g23_test tests/core_test.cpp)
 target_link_libraries(g23_test g23)
 add_executable(g23_lut_test tests/lut_test.cpp)
 target_link_libraries(g23_lut_test g23)
 add_test(NAME lut COMMAND g23_lut_test)
 add_test(NAME core COMMAND g23_test ${CMAKE_CURRENT_SOURCE_DIR}/../protocol/default_parameters.json)
endif()
